
N = 4

def print_solution(board):
    for row in board:
        print(" ".join("Q" if x else "." for x in row))
    print()


def is_safe(board, row, col):
    # Check upper column
    for i in range(row):
        if board[i][col] == 1:
            return False

    # Check upper-left diagonal
    i, j = row - 1, col - 1
    while i >= 0 and j >= 0:
        if board[i][j] == 1:
            return False
        i -= 1
        j -= 1

    # Check upper-right diagonal
    i, j = row - 1, col + 1
    while i >= 0 and j < N:
        if board[i][j] == 1:
            return False
        i -= 1
        j += 1

    return True


def solve_n_queens(board, row):
    if row == N:
        print_solution(board)
        return

    for col in range(N):
        if is_safe(board, row, col):
            board[row][col] = 1

            solve_n_queens(board, row + 1)

            # Backtracking
            board[row][col] = 0


# Main program
board = [[0 for _ in range(N)] for _ in range(N)]

print("All possible solutions for 4-Queens:")
solve_n_queens(board, 0)
