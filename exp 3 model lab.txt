% Rule to find students belonging to a department
department_student(Name, Dept) :-
    student(Name, Dept, _).

% Rule to find students above a given age
above_age(Name, Age) :-
    student(Name, _, StudentAge),
    StudentAge > Age.

% Rule to display student details
display_student(Name, Dept, Age) :-
    student(Name, Dept, Age).
